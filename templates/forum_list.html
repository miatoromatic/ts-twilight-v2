// ===== Activity feed: size controls (Hide / drag to set height / Default / Full) ===== // Starts at DEFAULT px (or shorter if there isn't enough content). The choice is remembered per browser. (function() { var box = document.querySelector('.tsActivity'); if (!box) { return; } var panes = box.querySelector('.tsActivity-panes'); var KEY = 'tsActivityView', MIN = 120, DEFAULT = 415; var state = {}; // no saved height yet = Default; 0 = Full (no limit) var h = (typeof state.h === 'number') ? state.h : DEFAULT; box.classList.toggle('is-sized', !!h); panes.style.maxHeight = h ? h + 'px' : ''; } apply(); box.addEventListener('click', function(e) { var btn = e.target.closest('[data-ts-size]'); if (!btn) { return; } var act = btn.getAttribute('data-ts-size'); if (act === 'hide') { state.hidden = true; } else if (act === 'show') { state.hidden = false; } else if (act === 'full') { state.h = 0; } else if (act === 'default') { state.h = DEFAULT; } else { return; } apply(); save(); }); // Drag the grip to set the height var grip = box.querySelector('.tsActivity-grip'); var startY = 0, startH = 0; function onMove(e) { var h = startH + (e.clientY - startY); h = Math.max(MIN, Math.min(h, panes.scrollHeight)); state.h = Math.round(h); apply(); } function onUp() { box.classList.remove('is-dragging'); document.removeEventListener('pointermove', onMove); document.removeEventListener('pointerup', onUp); if (state.h >= panes.scrollHeight) { state.h = 0; apply(); } // dragged to the bottom = full height save(); } grip.addEventListener('pointerdown', function(e) { e.preventDefault(); startY = e.clientY; startH = panes.getBoundingClientRect().height; box.classList.add('is-dragging'); document.addEventListener('pointermove', onMove);
