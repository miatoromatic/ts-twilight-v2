// Shop / News feed tabs: fetch the page the first time its tab is opened. // Follows XenForo's "what's new" redirect (e.g. whats-new/shop-items -> whats-new/shop-items/12345/). $(document).on('click', '.tsActivity-tab', function() { var $pane = $('#' + $(this).attr('aria-controls')); var $target = $pane.find('[data-ts-load]'); if (!$target.length || $target.data('tsLoaded')) { return; } $target.data('tsLoaded', true); var load = function(url, hops) { XF.ajax('get', url, {}, function(data) { if (data.redirect && hops < 3) { load(data.redirect, hops + 1); return; } if (data.html) { XF.setupHtmlInsert(data.html, function($html) { $target.html($html); }); } else { $target.html('
Nothing to show right now.
'); } }, { skipDefaultSuccess: true }).fail(function() { $target.data('tsLoaded', false); $target.html('
Couldn\'t load this tab. Use the link below.
'); }); }; load($target.data('ts-load'), 0); });
